Impact Calculations (Task 2):
  timeToMarketImpact = -(leadTimeChange) × dailyCost × throughput × months

  efficiencyImpact = -(wipChange) × costPerItem × months
  qualityImpact = -(defectsChange) × costPerItem × months

Data Persistence (Task 2):
  save: localStorage.setItem('key', JSON.stringify(data))
  load: JSON.parse(localStorage.getItem('key'))
  share: window.location.origin + '?' + new URLSearchParams(data)

Clipboard (Task 2):
  modern: navigator.clipboard.writeText(text)
  fallback: document.execCommand('copy')
